03 / SOURCE CODE

Complete, production-tested Arduino C++ sketch authored for the Acebott ESP32 Max V3.0 board. Built using the native <WiFi.h> and <WebServer.h> libraries with non-blocking HTTP 303 redirects.

Acebott_WebServer_LED.ino
/*
 * Acebott ESP32 Web Server LED Control
 * Institution: B.D.M. International (BDMI)
 * House: Courage House | Grade 8, Section 18
 * Board: Acebott ESP32 Max Controller Board
 * Pin: GPIO 32 (Connected to the light module)
 */

#include <WiFi.h>
#include <WebServer.h>

// --- Wi-Fi Credentials ---
const char* ssid = "YOUR_SSID";
const char* password = "YOUR_PASSWORD";

// --- Hardware Pin Definition ---
const int LED_PIN = 32;

// --- Initialize Web Server on Port 80 ---
WebServer server(80);

void setup() {
  // Initialize Serial Monitor at 115200 baud
  Serial.begin(115200);
  delay(1000);

  // Configure LED pin as digital output and ensure it starts LOW (Off)
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  // Connect to Wi-Fi network
  Serial.println();
  Serial.print("Connecting to ");
  Serial.println(ssid);

  WiFi.begin(ssid, password);

  // Wait for connection
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("");
  Serial.println("WiFi connected.");
  Serial.print("IP address: ");
  // Print the local IP address to the Serial Monitor
  Serial.println(WiFi.localIP());

  // Define HTTP route handlers
  server.on("/", handleRoot);
  server.on("/led/on", handleLedOn);
  server.on("/led/off", handleLedOff);

  // Start the server
  server.begin();
  Serial.println("HTTP server started");
}

void loop() {
  // Handle client requests
  server.handleClient();
}

// --- Route Handler: Root Dashboard ("/") ---
void handleRoot() {
  int ledState = digitalRead(LED_PIN);
  
  String html = "<html lang=\"en\">";
  html += "<head>";
  html += "<meta charset=\"UTF-8\">";
  html += "<meta name=\"viewport\" content=\"width=device-width, initial-scale=1.0\">";
  html += "<title>Acebott ESP32 Light Control</title>";
  html += "<style>";
  html += "body { font-family: Arial, sans-serif; text-align: center; background-color: #f4f4f9; margin-top: 50px; color: #333; }";
  html += ".container { display: inline-block; padding: 30px; background: white; border-radius: 10px; box-shadow: 0 4px 8px rgba(0,0,0,0.1); }";
  html += "h1 { color: #0066cc; }";
  html += ".status { font-size: 1.2rem; margin: 20px 0; font-weight: bold; }";
  html += ".btn { display: inline-block; padding: 12px 24px; font-size: 1rem; color: white; border: none; border-radius: 5px; text-decoration: none; cursor: pointer; margin: 10px; }";
  html += ".btn-on { background-color: #28a745; }";
  html += ".btn-off { background-color: #dc3545; }";
  html += ".btn:hover { opacity: 0.9; }";
  html += "</style>";
  html += "</head>";
  html += "<body>";
  html += "<div class=\"container\">";
  html += "<h1>Acebott ESP32 Light Control</h1>";
  
  if (ledState == HIGH) {
    html += "<p class=\"status\">Status: <span style=\"color: #28a745;\">ON</span></p>";
    html += "<a href=\"/led/off\" class=\"btn btn-off\">Turn OFF</a>";
  } else {
    html += "<p class=\"status\">Status: <span style=\"color: #dc3545;\">OFF</span></p>";
    html += "<a href=\"/led/on\" class=\"btn btn-on\">Turn ON</a>";
  }
  
  html += "</div>";
  html += "</body>";
  html += "</html>";

  server.send(200, "text/html", html);
}

// --- Route Handler: Turn LED ON ("/led/on") ---
void handleLedOn() {
  digitalWrite(LED_PIN, HIGH);
  // Redirect back to root dashboard with HTTP status 303 (See Other)
  server.sendHeader("Location", "/");
  server.send(303);
}

// --- Route Handler: Turn LED OFF ("/led/off") ---
void handleLedOff() {
  digitalWrite(LED_PIN, LOW);
  // Redirect back to root dashboard with HTTP status 303 (See Other)
  server.sendHeader("Location", "/");
  server.send(303);
}

Deployment Protocol

04 / WORKFLOW
01

Driver Installation

Ensure the CH340 USB driver is installed on Windows or macOS so that the virtual COM port enumerates correctly.

02

Board Environment Setup

In the Arduino IDE, set the target board to ESP32 Dev Module (Upload Speed: 921600, CPU Frequency: 240MHz, Flash Frequency: 80MHz).

03

Hardware Interfacing

Wire the Acebott Light Module Signal pin to GPIO 32, the VCC pin to 5V / 3.3V, and GND to GND on the header pins.

04

Firmware Flashing

Configure your 2.4 GHz network credentials in the sketch, hit compile/upload, and open the Serial Monitor at 115200 baud to fetch the assigned IP address.
